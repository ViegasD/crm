# ClickUp Feature Import — CRM2 Scope
# Structure: 22 Lists (P1/P2/P3 | Stage N — name) → Tasks, created inside a target Folder
# ClickUp does not support nested folders — lists are created directly inside the CRM2 folder.
#
# Usage: .\scripts\clickup_import.ps1 -ApiToken "pk_xxx" -FolderId "901318283724"
#    or set CLICKUP_API_TOKEN / CLICKUP_FOLDER_ID in scripts/.env
#
# How to find your Folder ID from the URL:
#   https://app.clickup.com/{teamId}/v/f/{FolderId}

param(
    [string]$ApiToken,
    [string]$FolderId
)

. (Join-Path $PSScriptRoot "clickup_common.ps1")

$config = Resolve-ClickUpConfig -ApiToken $ApiToken -FolderId $FolderId
$headers = $config.Headers
$FolderId = $config.FolderId

if (-not $FolderId) {
    throw "ClickUp folder id not provided. Set CLICKUP_FOLDER_ID in scripts/.env or pass -FolderId."
}

function New-List($folderId, $name) {
    $b = @{ name = $name } | ConvertTo-Json
    $r = Invoke-RestMethod -Method Post -Uri "https://api.clickup.com/api/v2/folder/$folderId/list" -Headers $headers -Body $b
    Write-Host "  LIST: $name ($($r.id))"
    return $r.id
}

function New-Task($listId, $name, $description) {
    $b = @{ name = $name; markdown_description = $description } | ConvertTo-Json -Depth 5
    Invoke-RestMethod -Method Post -Uri "https://api.clickup.com/api/v2/list/$listId/task" -Headers $headers -Body $b | Out-Null
    Write-Host "    TASK: $name"
}
